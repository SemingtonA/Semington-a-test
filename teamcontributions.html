    ${player} ${data.legsWon.toFixed(2)} ${data.spares.toFixed(2)} ${data.topScore.toFixed(2)} ${data.duckFines.toFixed(2)} ${data.legFines.toFixed(2)} ${data.leagueFees.toFixed(2)} ${data.cupFees.toFixed(2)} ${data.legFees.toFixed(2)} ${data.netOwed.toFixed(2)}

    `; document.getElementById("tableContainer").innerHTML = html; }
